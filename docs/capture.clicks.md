Both <form> submissions and <map> / <area> (usemap) clicks trigger standard browser navigations. [1] (https://developer.mozilla.org/en-US/docs/Web/API/NavigateEvent)
The cleanest way to capture both in one place is the Navigation API, with traditional event delegation as a fallback. [1] (https://developer.mozilla.org/en-US/docs/Web/API/Navigation/navigate_event), [2] (https://developer.mozilla.org/en-US/docs/Web/API/Navigation_API)

1. Unified Approach: Navigation API (navigate event)

Per the MDN NavigateEvent documentation, window.navigation intercepts <form> submissions (both GET and POST) as well as clicks on <area href="..."> elements inside an image map. [1] (https://developer.mozilla.org/en-US/docs/Web/API/NavigateEvent)
javascript
window.navigation.addEventListener("navigate", (event) => {
  // Ignore cross-origin navigations or file downloads (e.g., <area download>)
  if (!event.canIntercept || event.downloadRequest !== null) {
    return;
  }

  const destinationUrl = new URL(event.destination.url);

  event.intercept({
    async handler() {
      // 1. POST <form> submission: data is attached to event.formData
      if (event.formData) {
        console.log("POST form data:", Object.fromEntries(event.formData));
        // Send event.formData via fetch() and update UI
        return;
      }

      // 2. GET <form> submission: inputs are already serialized in destinationUrl.searchParams
      if (event.sourceElement?.closest("form")) {
        console.log("GET form params:", Object.fromEntries(destinationUrl.searchParams));
        // Render results for destinationUrl
        return;
      }

      // 3. <map> / usemap hit: sourceElement is the clicked <area>
      if (event.sourceElement instanceof HTMLAreaElement) {
        console.log("Usemap area clicked:", event.sourceElement.alt, destinationUrl.href);
        // Handle image map route change
        return;
      }

      // 4. All other same-origin navigations (<a> links, back/forward)
      console.log("Standard navigation to:", destinationUrl.href);
    },
  });
});
Use code with caution.

Key NavigateEvent Properties for Forms & Image Maps

• event.formData: Populated with a FormData instance for POST forms; null for GET forms and <area> clicks.
• event.destination.url: Contains the full target URL, including query strings generated by GET forms or <area href="..."> attributes.
• event.sourceElement: References the HTMLAreaElement when an image map region is clicked, or the <form> / submit <button> when a form is submitted. [1] (https://developer.mozilla.org/en-US/docs/Web/API/NavigateEvent/formData), [2] (https://developer.mozilla.org/en-US/docs/Web/API/NavigateEvent), [3] (https://developer.mozilla.org/en-US/docs/Web/API/Navigation/navigate_event)

If you need to support older browsers without window.navigation, intercept submit and click events at the document level and update the URL manually: [1] (https://developer.mozilla.org/en-US/docs/Web/API/HTMLFormElement/submit_event), [2] (https://developer.mozilla.org/en-US/docs/Learn_web_development/Extensions/Forms/User_input_methods), [3] (https://developer.mozilla.org/en-US/docs/Web/API/Navigation/navigate_event)

// Capture <form> submissions
document.addEventListener("submit", (event) => {
  const form = event.target;
  if (!(form instanceof HTMLFormElement)) return;

  event.preventDefault();
  const formData = new FormData(form, event.submitter);

  if (form.method.toLowerCase() === "get") {
    const url = new URL(form.action || window.location.href);
    url.search = new URLSearchParams(formData).toString();
    history.pushState(null, "", url);
    handleRouteChange(url);
  } else {
    // Handle POST submission via fetch(form.action, { method: "POST", body: formData })
  }
});

// Capture <map> / usemap <area> hits
document.addEventListener("click", (event) => {
  const area = event.target.closest("area[href]");
  if (!area) return;

  // Allow modified clicks (open in new tab) to behave normally
  if (event.metaKey || event.ctrlKey || event.shiftKey || event.altKey) return;

  const url = new URL(area.href, window.location.href);
  if (url.origin !== window.location.origin) return;

  event.preventDefault();
  history.pushState(null, "", url);
  handleRouteChange(url);
});

// Handle Back / Forward buttons
window.addEventListener("popstate", () => {
  handleRouteChange(new URL(window.location.href));
});

Server-side image maps encode click coordinates directly into the navigation target—either as ?x,y via <img ismap> or as x and y form fields via <input type="image">.

How Browsers Format the Coordinates


Element	Required Parent	Coordinate Format	Where Coordinates Appear
<img ismap>	<a href="...">	?120,85 (comma-separated CSS pixels)	Appended to destination.url (url.search)
<input type="image">	<form>	?x=120&y=85 (or ?name.x=120&name.y=85 if name is set)	destination.url (GET) or event.formData (POST)


1. Unified Capture: Navigation API (navigate event)

Because the browser computes the click coordinates before dispatching navigate, event.destination.url and event.formData already contain the exact values:
javascript
window.navigation.addEventListener("navigate", (event) => {
  if (!event.canIntercept || event.downloadRequest !== null) return;

  const url = new URL(event.destination.url);

  event.intercept({
    async handler() {
      // Case 1: <a href="/map"><img ismap></a>
      // url.search is formatted as "?x,y" (e.g., "?120,85")
      if (event.sourceElement?.querySelector("img[ismap]")) {
        const [x, y] = url.search.slice(1).split(",").map(Number);
        console.log("ismap hit at:", { x, y, href: url.pathname });
        return;
      }

      // Case 2: <form> submitted via <input type="image" name="map">
      // Coordinates are in formData (POST) or url.searchParams (GET)
      const params = event.formData ?? url.searchParams;
      const xKey = [...params.keys()].find((k) => k === "x" || k.endsWith(".x"));
      const yKey = [...params.keys()].find((k) => k === "y" || k.endsWith(".y"));

      if (xKey && yKey) {
        const x = Number(params.get(xKey));
        const y = Number(params.get(yKey));
        console.log("input[type=image] hit at:", { x, y, prefix: xKey.replace(/\.?x$/, "") });
        return;
      }
    },
  });
});
Use code with caution.

2. Legacy Fallback: Event Delegation + History API

If you call event.preventDefault() on click or submit in older browsers, the browser will not automatically append the coordinates to the URL or FormData. You must calculate the CSS pixel offset from the image's top-left corner manually using getBoundingClientRect():
javascript
// 1. Intercept <a href="..."><img ismap></a>
document.addEventListener("click", (event) => {
  const img = event.target.closest("a[href] img[ismap]");
  if (!img) return;
  if (event.metaKey || event.ctrlKey || event.shiftKey || event.altKey) return;

  const anchor = img.closest("a[href]");
  const url = new URL(anchor.href, window.location.href);
  if (url.origin !== window.location.origin) return;

  event.preventDefault();

  const rect = img.getBoundingClientRect();
  const x = Math.round(event.clientX - rect.left);
  const y = Math.round(event.clientY - rect.top);

  url.search = `?${x},${y}`;
  history.pushState({ x, y }, "", url);
  handleMapHit({ x, y, url });
});

// 2. Intercept <form> submitted with <input type="image">
document.addEventListener("click", (event) => {
  const imageInput = event.target.closest('input[type="image"]');
  if (!imageInput || !imageInput.form) return;

  const rect = imageInput.getBoundingClientRect();
  // Store coordinates on the element so the subsequent "submit" listener can read them
  imageInput.dataset.clickX = Math.round(event.clientX - rect.left);
  imageInput.dataset.clickY = Math.round(event.clientY - rect.top);
});

document.addEventListener("submit", (event) => {
  const submitter = event.submitter;
  if (!(submitter instanceof HTMLInputElement) || submitter.type !== "image") return;

  event.preventDefault();
  const form = event.target;
  const formData = new FormData(form);
  const prefix = submitter.name ? `${submitter.name}.` : "";

  formData.set(`${prefix}x`, submitter.dataset.clickX ?? "0");
  formData.set(`${prefix}y`, submitter.dataset.clickY ?? "0");

  if (form.method.toLowerCase() === "get") {
    const url = new URL(form.action || window.location.href);
    url.search = new URLSearchParams(formData).toString();
    history.pushState(null, "", url);
    handleMapHit({ url, formData });
  }
});
Use code with caution.
(Note: Keyboard activation of <input type="image"> via Enter or Space submits x=0 and y=0 by specification.) [1] (https://funwithforms.com/posts/input-type-image/)
